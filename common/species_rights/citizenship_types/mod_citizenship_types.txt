citizenship_assimilation = {
	is_assimilation = yes
	potential = {
		exists = from
		from = {
			OR = {
				is_homicidal = no
				AND = {
					is_homicidal = yes
					root = {
						is_same_species = prev
					}
				}
				AND = {
					has_civic = civic_machine_terminator
					root = {
						is_robotic_species = yes
					}
				}
			}
			OR = {
				AND = {
					# machine cyborg assimilator
					is_machine_empire = yes
					has_civic = civic_machine_assimilator
					OR = {
						AND = {
							root = {
								is_robotic_species = yes
							}
						}
						root = {
							NOR = {
								has_trait = trait_mechanical
								has_trait = trait_machine_unit
								has_trait = trait_cybernetic
							}
						}
					}
				}
				AND = {
					# machine synthetic assimilator
					is_machine_empire = yes
					root = {
						is_robotic_species = yes
					}
				}
				AND = {
					OR = {
						AND = {
							# non-devouring hive mind
							is_hive_empire = yes
							NOT = {
								has_civic = civic_hive_devouring_swarm
							}
						}
						# Regular empire
						AND = {
							NOT = {
								has_ethic = ethic_gestalt_consciousness
							}
							is_individual_machine = no
						}
					}
					OR = {
						can_perform_genetic_assimilation = yes
						has_active_tradition = tr_cybernetics_transubstantiation_synthesis						# cybernetic assimilation
					}
					ROOT = {
						is_robotic_species = no
					}
				}
				AND = {
					has_country_flag = synth_assimilation					# synth assimilation
					is_machine_empire = no
				}
				AND = {
					OR = {
						has_active_tradition = tr_psionics_psionic_assimilation						# psionic assimilation
						has_active_tradition = tr_psionics_shroud_great_awakening						# psionic assimilation (shroud)
					}
					is_machine_empire = no
					is_individual_machine = no
					ROOT = {
						is_robotic_species = no
					}
				}
				AND = {
					is_individual_machine = yes
					ROOT = {
						is_robotic_species = yes
					}
				}
				AND = {
					# Mindwarden Anti-Psionics
					has_modifier = destroyers_of_the_lost_modifier
					ROOT = {
						has_psionic_species_trait = yes
					}
				}
			}
		}
	}
	allow = {
		hidden_trigger = {
			exists = from
		}
		custom_tooltip = {
			fail_text = MUST_NOT_BE_DOMINANT_SPECIES
			NOT = {
				is_exact_same_species = from
			}
		}
		custom_tooltip = {
			fail_text = "pre_sapients_citizenship"
			is_sapient = yes
		}
		custom_tooltip = {
			fail_text = "WILDERNESS_DEAD_WORLD"
			if = {
				limit = {
					from = {
						is_wilderness_empire = no
					}
				}
				NOT = {
					has_trait = trait_wilderness
				}
			}
		}
		custom_tooltip = {
			fail_text = "lonely_hive_mind_pops_kill_themselves_citizenship_assimilation"
			if = {
				limit = {
					is_lonely_hive_mind_pop = yes
				}
				from = {
					OR = {
						can_perform_genetic_assimilation = yes
						has_active_tradition = tr_cybernetics_transubstantiation_synthesis
						has_active_tradition = tr_cybernetics_assimilator_inevitable_assimilation
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = "assimilate_shroud_forged_tt"
			NOT = {
				has_trait = trait_shroud_forged
			}
		}
		OR = {
			custom_tooltip = {
				fail_text = "disconnected_machine_pops_collapse_citizenship"
				OR = {
					from = {
						has_country_flag = synth_assimilation
					}
					is_disconnected_machine_pop = no
				}
			}
			if = {
				limit = {
					from = {
						is_machine_empire = yes
					}
				}
				custom_tooltip = {
					# machine assimilator
					fail_text = MACHINE_SPECIES_MUST_BE_ORGANIC
				}
				custom_tooltip = {
					fail_text = requires_tradition_tr_synthetics_adopt
				}
			}
			if = {
				limit = {
					from = {
						is_hive_empire = yes
						is_wilderness_empire = no
						can_perform_genetic_assimilation = yes
					}
				}
				custom_tooltip = {
					# non-devouring hive mind
					fail_text = HIVE_SPECIES_MUST_BE_NON_HIVE
					NOR = {
						has_trait = trait_hive_mind
						has_trait = trait_mechanical
						has_trait = trait_machine_unit
					}
				}
			}
			if = {
				limit = {
					from = {
						is_wilderness_empire = yes
						can_perform_genetic_assimilation = yes
					}
				}
				custom_tooltip = {
					# non-devouring hive mind
					fail_text = HIVE_SPECIES_MUST_BE_ORGANIC
					NOR = {
						has_trait = trait_mechanical
						has_trait = trait_machine_unit
						has_trait = trait_wilderness
					}
				}
			}
			if = {
				limit = {
					from = {
						is_hive_empire = yes
						has_active_tradition = tr_cybernetics_transubstantiation_synthesis
					}
				}
				custom_tooltip = {
					# non-devouring hive mind
					fail_text = HIVE_SPECIES_MUST_BE_NON_HIVE
					NOR = {
						has_trait = trait_mechanical
						has_trait = trait_machine_unit
						AND = {
							has_trait = trait_hive_mind
							has_trait = trait_cybernetic
						}
					}
				}
			}
			if = {
				limit = {
					from = {
						is_hive_empire = no
						has_active_tradition = tr_cybernetics_transubstantiation_synthesis
					}
				}
				custom_tooltip = {
					# cyborg empire
					fail_text = CYBORG_SPECIES_MUST_BE_ORGANIC
					OR = {
						AND = {
							has_trait = trait_hive_mind
							from = {
								NOT = {
									is_hive_empire = yes
								}
							}
						}
						AND = {
							NOT = {
								has_trait = trait_hive_mind
							}
							from = {
								is_hive_empire = yes
							}
						}
						NOR = {
							has_trait = trait_mechanical
							has_trait = trait_machine_unit
							has_trait = trait_cybernetic
							has_trait = trait_limited_cybernetic
						}
					}
				}
			}
			if = {
				limit = {
					from = {
						has_active_tradition = tr_psionics_psionic_assimilation
					}
				}
				custom_tooltip = {
					# species must be organic to transcend
					fail_text = TRANSCENDING_SPECIES_MUST_BE_ORGANIC
					is_robotic_species = no
				}
				custom_tooltip = {
					# species already has psionic trait
					fail_text = TRANSCENDING_SPECIES_IS_ALREADY_PSIONIC
					is_psionic_species = no
				}
				custom_tooltip = {
					# transcending species cannot be gestalt consciousness
					fail_text = TRANSCENDING_SPECIES_CANT_BE_HIVE
					NOT = {
						has_trait = trait_hive_mind
					}
				}
			}
			if = {
				limit = {
					from = {
						has_active_tradition = tr_psionics_shroud_great_awakening
					}
				}
				# non organic and gestalt species are allowed in Shroud
				custom_tooltip = {
					# species already has psionic trait
					fail_text = TRANSCENDING_SPECIES_IS_ALREADY_PSIONIC
					is_psionic_species = no
				}
			}
			if = {
				limit = {
					from = {
						can_perform_genetic_assimilation = yes
					}
				}
				custom_tooltip = {
					fail_text = CAN_ONLY_ASSIMILATE_HIVES
					has_trait = trait_hive_mind
				}
			}
			if = {
				limit = {
					from = {
						can_perform_genetic_assimilation = yes
					}
				}
				custom_tooltip = {
					fail_text = MUST_NOT_BE_DOMINANT_SPECIES
					NOT = {
						is_exact_same_species = from
					}
				}
			}
		}
	}
	ai_will_do = {
		factor = 100
		modifier = {
			factor = 0
			from = {
				OR = {
					has_valid_civic = civic_hive_devouring_swarm
					has_valid_civic = civic_fanatic_purifiers
					has_valid_civic = civic_machine_terminator
				}
			}
		}
		modifier = {
			factor = 0
			has_species_flag = original_main_species
			from = {
				has_origin = origin_shroud_forged
			}
		}
		modifier = {
			factor = 0.3
			from = {
				is_individual_machine = yes
			}
		}
	}
}

citizenship_purge = {
	new_pop_weight = 0
	is_purge = yes
	potential = {
		exists = from
		NOT = {
			from = {
				is_machine_empire = yes
			}
		}
		OR = {
			NOT = {
				has_trait = trait_mechanical
				has_trait = trait_machine_unit
			}
			from = {
				has_policy_flag = robots_outlawed
			}
			AND = {
				from = {
					is_hive_empire = yes
				}
				custom_tooltip = {
					fail_text = PURGING_NOT_ALLOWED
					from = {
						NOT = {
							has_policy_flag = purge_not_allowed
						}
					}
				}
			}
		}
	}
	allow = {
		hidden_trigger = {
			exists = from
		}
		if = {
			limit = {
				is_lonely_hive_mind_pop = yes
			}
			hidden_trigger = {
				always = yes
			}
		}
		else_if = {
			limit = {
				is_disconnected_drone_pop = yes
			}
			from = {
				has_policy_flag = disconnected_drones_purge
			}
		}
		else = {
			custom_tooltip = {
				fail_text = is_broken_shackles_origin
				if = {
					limit = {
						OR = {
							is_robotic = no
							AND = {
								is_robotic = yes
								NOT = {
									from = {
										has_policy_flag = robots_outlawed
									}
								}
							}
						}
					}
					from = {
						NOT = {
							has_origin = origin_broken_shackles
						}
					}
				}
			}
			custom_tooltip = {
				fail_text = "presapients_not_purged_citizenship"
				OR = {
					is_sapient = yes
					from = {
						NOT = {
							has_policy_flag = pre_sapients_protect
							has_policy_flag = pre_sapients_allow
							has_policy_flag = pre_sapients_eat
						}
					}
				}
			}
			if = {
				limit = {
					NOR = {
						has_trait = trait_machine_unit
						has_trait = trait_mechanical
					}
				}
				custom_tooltip = {
					fail_text = PURGING_NOT_ALLOWED
					from = {
						NOT = {
							has_policy_flag = purge_not_allowed
						}
					}
				}
			}
			if = {
				limit = {
					from = {
						is_hive_empire = yes
					}
					has_trait = trait_hive_mind
				}
				custom_tooltip = {
					fail_text = hive_cant_purge_hive
					has_genesis_preserve_trait = no
				}
			}
			custom_tooltip = {
				fail_text = yes_service_yes_citizenship
				if = {
					limit = {
						from = {
							has_valid_civic = civic_citizen_service
						}
					}
					NOT = {
						has_military_service_type = {
							type = military_service_full
							country = from
						}
					}
				}
			}
			custom_tooltip = {
				fail_text = selective_kinship_must_citizen
				if = {
					limit = {
						from = {
							has_valid_civic = civic_selective_kinship
						}
					}
					NOT = {
						is_same_species_class = from
					}
				}
			}
		}
	}
	ai_will_do = {
		factor = 20
		modifier = {
			factor = 0
			is_same_species = from
		}
		modifier = {
			factor = 0
			has_trait = trait_machine_unit
			from = {
				is_machine_empire = yes
			}
		}
		modifier = {
			factor = 0
			from = {
				has_valid_ai_personality = no
				is_hive_empire = no
			}
		}
		modifier = {
			factor = 0
			from = {
				has_valid_ai_personality = yes
				NOR = {
					has_ai_personality_behaviour = purger
					AND = {
						has_ai_personality_behaviour = displacer
						if = {
							limit = {
								has_origin = origin_necrophage
							}
							is_hive_empire = yes
						}
					}
				}
			}
		}
		modifier = {
			factor = 0
			from = {
				is_machine_empire = yes
				has_government = gov_machine_industrial
			}
		}
		modifier = {
			factor = 0
			from = {
				is_hive_empire = yes
				NOT = {
					has_valid_civic = civic_hive_devouring_swarm
				}
				NOT = {
					has_origin = origin_necrophage
				}
				OR = {
					has_valid_civic = civic_hive_one_mind
					has_valid_civic = civic_hive_subsumed_will
					root = {
						has_trait = trait_hive_mind
					}
				}
			}
		}
		modifier = {
			factor = 0
			from = {
				has_origin = origin_syncretic_evolution
			}
		}
		modifier = {
			factor = 10
			from = {
				is_ai = yes
				has_ascension_perk = ap_become_the_crisis
			}
		}
		modifier = {
			add = 1000
			not = {
				has_species_flag = syncretic_species@from
			}
			is_robotic = no
			from = {
				has_origin = origin_necrophage
				is_xenophobe = yes
			}
		}
	}
}
